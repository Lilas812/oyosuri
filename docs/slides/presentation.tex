% =================  2. 通常のランダムウォーク  =====================
\begin{frame}[t]{通常のランダムウォーク：定義と確率分布}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      {\color{myblue}\bfseries 定義}
      \begin{itemize}\footnotesize
        \item 各ステップ $T_k$ は\alert{独立}
          \\ $P(T_k{=}{-}1){=}p,\ P(T_k{=}{+}1){=}q$（$p{+}q{=}1$）
        \item $n$ 歩後の位置
          \\ $\displaystyle S_n=\sum_{k=1}^{n}T_k$（$S_0{=}0$）
        \item 直前の向きは次に\alert{無関係}
        \item 分布は\alert{二項係数}で決まる
          \[ P(S_n{=}x)=\binom{n}{\ell}\,p^{\ell}q^{m},\quad
             \ell{=}\tfrac{n-x}{2},\ m{=}\tfrac{n+x}{2} \]
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \makebox[\linewidth][l]{{\color{myblue}\bfseries $n=1,2,3$ の確率分布}}
      \vskip0.4em
      {\footnotesize
      \renewcommand{\arraystretch}{1.45}
      \begin{tabular}{@{}c@{\ \ }l@{}}
        \hline
        $n$ & 位置ごとの確率 \\ \hline
        $1$ & $P({-}1){=}p,\ \ P({+}1){=}q$ \\
        $2$ & $P({-}2){=}p^{2},\ P(0){=}2pq,\ P({+}2){=}q^{2}$ \\
        $3$ & $P({-}3){=}p^{3},\ \ P({-}1){=}3p^{2}q,$ \\
            & $P({+}1){=}3pq^{2},\ P({+}3){=}q^{3}$ \\ \hline
      \end{tabular}}
      \vskip0.6em
      {\footnotesize\color{mygray}例：$P(S_2{=}0){=}2pq$ の係数 $2$ は，
      「左右1回ずつ」が左右・右左の\alert{2通り}あることを表す（順序によらない）}
    \end{column}
  \end{columns}
\end{frame}
\note{% 【約50秒】
まず通常のランダムウォークです。各ステップ $T$ は独立に，確率 $p$ でマイナス1，確率 $q$ でプラス1動きます。$S_n$ は，原点から $n$ 回移動した後の位置で，各ステップを累積した値です。直前の向きは次に影響しないため，時刻 $n$ で位置 $x$ にいる確率は二項係数で決まり，左右の移動の順序にはよりません。右の表が $n=1,2,3$ の分布です。たとえば $n=2$ で原点にいる確率は $2pq$。これは左右1回ずつの経路が，左右・右左の2通りあることを表す係数2をもちます。この「順序によらない」点が，次の相関付きとの違いになります。
\sourceblock{docs/thesis/main.tex, 第3章「ランダムウォークについて」。}%
}

% =================  3. 相関付きRW（定義・行列定式化）  ============
\begin{frame}[t]{相関付きランダムウォーク：定義と行列による定式化}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \hspace*{1em}{\color{myblue}\bfseries 定義}：直前の向きが次に\alert{影響}
      \begin{itemize}\footnotesize
        \item $P(\text{左}\mid\text{直前左}){=}a,\ P(\text{右}\mid\text{直前左}){=}c{=}1{-}a$
        \item $P(\text{左}\mid\text{直前右}){=}b,\ P(\text{右}\mid\text{直前右}){=}d{=}1{-}b$
      \end{itemize}
      \vskip0.3em
      コイン行列と\alert{左右移動への分解}
      \[ A=\begin{pmatrix}a&b\\c&d\end{pmatrix}
         =\underbrace{\begin{pmatrix}a&b\\0&0\end{pmatrix}}_{P\ (\text{左移動})}
          +\underbrace{\begin{pmatrix}0&0\\c&d\end{pmatrix}}_{Q\ (\text{右移動})} \]
    \end{column}
    \begin{column}{0.49\textwidth}
      {\color{myblue}\bfseries 確率ベクトルの時間発展}
      \begin{itemize}\footnotesize
        \item 時刻 $n$・位置 $x$ の確率ベクトルと初期値
          \[ \Psi_n(x)=\begin{pmatrix}\Psi_n^{L}(x)\\[1pt]\Psi_n^{R}(x)\end{pmatrix},\ \
             \Psi_0(0)=\varphi=\begin{pmatrix}\alpha\\\beta\end{pmatrix} \]
        \item 発展則（$P$ が左へ，$Q$ が右へ運ぶ）
          \[ \Psi_{n+1}(x)=P\,\Psi_n(x{+}1)+Q\,\Psi_n(x{-}1) \]
        \item 位置の確率は成分の和
          \[ P(\tilde S_n{=}x)=\lVert\Psi_n(x)\rVert_1=\Psi_n^{L}(x)+\Psi_n^{R}(x) \]
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
\note{% 【約45秒】
相関付きランダムウォークです。直前が左なら次に左へ行く確率を $a$，右なら右へ行く確率を $d$ などと定め，コイン行列 $A$ にまとめます。ここが本質で，$A$ を左へ動かす成分 $P$ と右へ動かす成分 $Q$ に分解します。すると，時刻 $n$・位置 $x$ の状態を確率ベクトル $\Psi$ で表し，$P$ と $Q$ を掛けながら1歩ずつ時間発展させることで分布が計算できます。位置の確率は，そのベクトルの成分の和です。
\sourceblock{docs/thesis/main.tex, 第4章「相関付きランダムウォークについて」。}%
}

% =================  4. 相関付きRW（行列計算 n=1,2）  ==============
\begin{frame}[t]{相関付きランダムウォーク：行列計算（$n=1,2$）}
  \small
  {\color{myblue}\bfseries $n=1$（原点から1歩目）}：初期ベクトル $\varphi$ に $P,Q$ を掛ける
  {\footnotesize
  \[\begin{aligned}
    \Psi_1(-1)&=P\varphi=\begin{pmatrix}a&b\\0&0\end{pmatrix}\begin{pmatrix}\alpha\\\beta\end{pmatrix}
      =\begin{pmatrix}a\alpha+b\beta\\0\end{pmatrix},\qquad
    \Psi_1(+1)=Q\varphi=\begin{pmatrix}0&0\\c&d\end{pmatrix}\begin{pmatrix}\alpha\\\beta\end{pmatrix}
      =\begin{pmatrix}0\\c\alpha+d\beta\end{pmatrix}
  \end{aligned}\]}
  {\color{myblue}\bfseries $n=2$（発展則 $\Psi_{n+1}(x){=}P\Psi_n(x{+}1){+}Q\Psi_n(x{-}1)$ を適用）}
  {\footnotesize
  \[\begin{aligned}
    \Psi_2(-2)&=P\Psi_1(-1)=\begin{pmatrix}a(a\alpha+b\beta)\\0\end{pmatrix},\qquad
    \Psi_2(+2)=Q\Psi_1(+1)=\begin{pmatrix}0\\d(c\alpha+d\beta)\end{pmatrix}\\[2pt]
    \Psi_2(0)&=P\Psi_1(+1)+Q\Psi_1(-1)
      =\begin{pmatrix}b(c\alpha+d\beta)\\0\end{pmatrix}+\begin{pmatrix}0\\c(a\alpha+b\beta)\end{pmatrix}
      =\begin{pmatrix}b(c\alpha+d\beta)\\c(a\alpha+b\beta)\end{pmatrix}
  \end{aligned}\]}
  \centering{\footnotesize\color{mygray}位置の確率は成分の和：$P(\tilde S_2{=}0)=\lVert\Psi_2(0)\rVert_1=b(c\alpha+d\beta)+c(a\alpha+b\beta)$}
\end{frame}
\note{% 【約55秒】
では実際に行列で計算します。1歩目は，初期ベクトル $\varphi$ に $P$ と $Q$ を掛けるだけです。$P$ を掛けると左向き成分に $a\alpha{+}b\beta$，$Q$ を掛けると右向き成分に $c\alpha{+}d\beta$ が現れ，これがそれぞれ位置マイナス1，プラス1にいる確率です。2歩目は，発展則にしたがって1歩目のベクトルにまた $P$ と $Q$ を掛けて足します。たとえば原点に戻る $\Psi_2(0)$ は，右にいた状態に $P$ を掛けたものと，左にいた状態に $Q$ を掛けたものの和で，$b(c\alpha{+}d\beta){+}c(a\alpha{+}b\beta)$ になります。このように，行列を掛けて足すだけで分布が求まります。$n=3$ は経路順序ごとの行列積が増えるため，発表では省略します。
\sourceblock{docs/thesis/main.tex, 第4章の \(n=1,2\) の時間発展および経路和。}%
}

% =================  5. 予測モデル（評価関数・当てはめ）  ==========
\begin{frame}[t]{\hspace*{0.1em}相関付きRWによる時系列予測：評価関数と当てはめ}
  \small
  観測時系列 $x_0,\dots,x_n$（増分 $\pm1$，$x_0{=}0$）に対し，パラメータ
  $(p,q,\alpha)$ を推定して次時刻を予測する。
  \par{\footnotesize\color{mygray}$p,q$：左右の継続確率，$\alpha$：初期が左向きの確率}
  \vskip0.4em
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      {\color{myblue}\bfseries ① モデル分布とのズレを測る}
      \begin{itemize}\footnotesize
        \item $\mu_t(x;p,q,\alpha){=}P(\tilde S_t{=}x)$
          \\ ＝\alert{発展則で求めるモデルの分布}そのもの
        \item 評価関数（二乗誤差を確率で重み付け）
          {\footnotesize
          \[V_n(p,q,\alpha){=}\sum_{t=0}^{n}\sum_{x}(x{-}x_t)^{2}\,\mu_t(x;p,q,\alpha)\]}
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      {\color{myblue}\bfseries ② パラメータを決めて予測する}
      \begin{itemize}\footnotesize
        \item 評価関数 $V_n$ が\alert{最小となるパラメータ}
          \\ $(p^\ast,q^\ast,\alpha^\ast)$ を決める
        \item 次時刻を\alert{期待値}で予測
          {\footnotesize
          \[x_{n+1}^{\ast}{=}E(\tilde S_{n+1}\mid p^\ast,q^\ast,\alpha^\ast){=}\sum_x x\,\mu_{n+1}(x;\!\cdot)\]}
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.2em
  \centering{\footnotesize\color{mygray}以上を各 $n$ で繰り返し，予測列 $x_1^{\ast},\dots,x_N^{\ast}$ を得る}
\end{frame}
\note{% 【約55秒】
では，この分布をどう時系列予測に使うかです。観測した増分プラスマイナス1の系列に対し，パラメータ $p$，$q$，$\alpha$ を推定します。左の $\mu$ が，先ほどの発展則を必要な時刻まで繰り返して求めるモデルの分布です。各時刻で，観測値 $x_t$ と，モデルが取りうる位置 $x$ の二乗誤差を，確率 $\mu$ で重み付けして足したものが評価関数 $V_n$ です。この評価関数が最小となるパラメータを決め，そのパラメータから得られる次時刻の期待値を予測値とします。これを各時刻で繰り返して，予測列を得ます。
\sourceblock{docs/thesis/main.tex, 第5章「時系列モデル」。src/oyosuri/model.py・optimize.py・predict.py。}%
}

      \centering
      {\color{myblue}\bfseries 変換後：平均練行足 $x_n$}\par
      \vskip0.25em
      \safeincludegraphics[width=\linewidth,height=31mm,keepaspectratio]{figures/mean-renko-after}
    \end{column}
  \end{columns}
  \vskip-0.05em
  \centering{\scriptsize\color{sub}価格系列から平均練行足への変換例}\par
  \vskip0pt
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.62\textwidth}
      {\color{myblue}\bfseries 変換規則}
      \begin{itemize}\footnotesize
        \item 基準価格 $o_i$ から上へ $B$ ドル：$b_i=+1$，下へ $B$ ドル：$b_i=-1$
        \item 価格が $B$ 動くごとに1本生成し，基準価格を更新
        \item 符号を累積して $\displaystyle x_n=\sum_{k=1}^{n}b_k$ を得る
      \end{itemize}
    \end{column}
    \begin{column}{0.36\textwidth}
      {\color{myblue}\bfseries 主実験}
      \begin{itemize}\footnotesize
        \item XAUUSD 1分足：317本
        \item $B=3$ ドル
        \item 平均練行足：$N=156$ 本
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
\note{% 【約45秒】
実際の金価格は連続的に変動するので，そのままではモデルに入れられません。左が変換前の価格系列，右が変換後の平均練行足です。基準価格から $B$ 以上上がればプラス1の足，下がればマイナス1の足とし，価格が $B$ 動くごとに1本生成します。こうして得た符号を累積すると，ボックス単位の整数時系列 $x_n$ になります。主実験ではXAUUSDの1分足317本に対して $B=3$ ドルとし，156本の平均練行足を得ました。この離散系列をモデルの入力として使います。

% =================  6. 実験：全期間の失敗と参照期間W  =============
\begin{frame}[t]{実験：金価格 XAUUSD への適用}
  \small
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      {\footnotesize {\color{myblue}\bfseries 設定}：XAUUSD 1分足（$B{=}3$ ドル，$N{=}156$）。予測と実測の\alert{重ね描き}・MAE・HIT で評価（変化なし予測：MAE$\,{=}1$，無作為な方向予測：HIT$\,{=}0.5$）。}
      \vskip0.4em
      \begin{itemize}\footnotesize
        \item \textbf{全期間参照}では\alert{誤差が大きい}
          \\ $\mathrm{MAE}{=}5.94$（基準の約6倍），$\mathrm{HIT}{=}0.43$
          \\ 原因：\alert{非定常}な値動きを定常と仮定
        \item 改良：当てはめを\alert{直近 $W$ 点}に限定（参照期間 $W$）
        \item $W$ を小さくすると
          \\ \alert{MAE・計算時間がともに減少}
          \\ $W{=}2$：MAE $0.847$（基準 $1$ 未満は $W{=}2,3$）
      \end{itemize}
    \end{column}
    \begin{column}{0.54\textwidth}
      \centering
      {\color{myblue}\bfseries 全期間参照の予測結果}\par
      \vskip0.25em
      \safeincludegraphics[width=\linewidth,height=48mm,keepaspectratio]{figures/full-history-overlay}
      \par{\footnotesize\color{mygray}実測値と全期間参照による予測値の重ね描き}
    \end{column}
  \end{columns}
\end{frame}
\note{% 【約52秒】
実験です。対象は金価格 XAUUSD の1分足，値幅3ドルで作った練行足156本。予測と実測の重ね描きを中心に，MAEと方向的中率で評価します。練行足は毎回プラスマイナス1動くので，変化なし予測のMAEは1，無作為な方向予測のHITは0.5が目安です。右図が，全期間を参照して得た予測値と実測値の重ね描きです。全期間参照ではMAEは5.94，HITは0.43となり，誤差が大きくなりました。値動きは上昇・下降・レンジが移り変わる非定常な系列なのに，全期間参照はこれを定常と仮定してしまうためです。そこで当てはめを直近 $W$ 点に限定すると，$W$ を小さくするほどMAEと計算時間がともに減少し，$W=2$ でMAEは最小の0.847となりました。

% =====================  8. モデルの限界  ==========================
\begin{frame}[t]{モデルの限界：局面転換への追従の遅れ}
  \small
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \centering
      \safeincludegraphics[width=\linewidth]{figures/pat_up2down_w}
      \par{\footnotesize\color{mygray}上昇→下降の局面（合成データ，$N=150$）}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{itemize}\footnotesize
        \item 大きい $W{=}60$ は転換点を過ぎても上昇の勢いを引きずり，実測から大きく乖離
          （\alert{$\mathrm{MAE}\,4.876$} vs $W{=}2$ の $0.892$）
        \item 機構：予測は\alert{参照期間の先頭から $W$ 歩後の期待値}であり，\alert{現在位置・直前の向きを条件づけない}
          \begin{itemize}\scriptsize
            \item $W$ 大 → 起点が遠く誤差増，かつ転換に $W$ 本の遅れ
          \end{itemize}
        \item \alert{「$W{=}2$ が最良」は精度の証明ではない}：履歴を使うほど誤差を上乗せする構造の下で，履歴をほぼ使わず悪化を最小化しただけ
        \item HIT は $W$ によらず約 $0.5$ → 方向を安定して当ててはいない
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
\note{% 【約55秒】
しかし，この結果はモデルの優秀さを意味しません。図は，上昇から下降へ転じる局面です。参照期間の大きい $W=60$ は，転換点を過ぎても上昇の勢いを引きずり，実測から大きく乖離して，MAEは4.876，$W=2$ の0.892の5倍以上に悪化しています。原因はモデルの構造です。この予測は，参照期間の先頭から $W$ 歩後の期待値を求めるだけで，現在位置や直前の向きで条件づけていません。だから $W$ が大きいほど，起点が現在から遠くなって誤差が増え，局面が変わっても $W$ 本ぶんの遅れが出ます。つまり，$W=2$ が最良なのは，履歴を使うほど誤差を上乗せするこの構造のもとで，履歴をほとんど使わないことで，悪化を最小に抑えただけ，と見るべきです。実際，方向的中率HITは $W$ によらずおよそ0.5にとどまり，次の向きを安定して当てられてはいません。
\sourceblock{docs/thesis/main.tex, 第8章「考察」。図は figures/pat\_up2down\_w.png。}%
}

% =====================  9. まとめ  ================================
\begin{frame}[t]{まとめと今後の課題}
  \small
  \begin{block}{まとめ}
    \begin{itemize}
      \item 相関付きランダムウォークを，金価格 XAUUSD に\alert{平均練行足}を通して初めて適用
      \item 全期間参照では誤差が大きい（$\mathrm{MAE}=5.94$）。\alert{参照期間 $W$} の導入で改善し，$W=2$ で最良（$\mathrm{MAE}=0.847$）
      \item \alert{現在位置・向きを予測に使わないことが限界}であり，$W$ を大きくすると誤差が増える
    \end{itemize}
  \end{block}
  \begin{block}{今後の課題}
    \begin{itemize}\footnotesize
      \item 現在位置・直前の向きを\alert{条件づける}予測への拡張（条件付き期待値）
      \item 金以外の資産（株・為替・暗号資産・他商品）への適用
      \item 平均練行足に依らない離散化・非一定増分への拡張
      \item マルコフ連鎖・高次モデル・レジーム転換モデルとの比較
    \end{itemize}
  \end{block}
\end{frame}
\note{% 【約45秒】
まとめます。本研究では，相関付きランダムウォークによる時系列モデルを，金価格に，平均練行足を通して初めて適用しました。全期間参照ではMAEが5.94と誤差が大きくなりましたが，参照期間 $W$ の導入で改善し，$W=2$ で最良の0.847が得られました。ただし，現在位置や向きを予測に使わないことが本モデルの限界であり，$W$ を大きくすると誤差が増えます。今後の課題としては，現在位置と直前の向きを条件づける予測への拡張，金以外の資産への適用，平均練行足に依らない離散化への拡張，そして，マルコフ連鎖など他のモデルとの比較が挙げられます。以上です。
\sourceblock{docs/thesis/main.tex, 第8章「考察」および第9章「今後の課題」。}%

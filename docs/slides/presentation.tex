% =================  2. 相関付きRW（定義・行列定式化）  ============
\begin{frame}[t]{相関付きランダムウォーク：定義と行列による定式化}
  \small
  {\footnotesize\color{mygray}通常のランダムウォークは各ステップが\alert{独立}（直前の向きは次に無関係）。
  ここでは直前の向きで確率を変える。}
  \vskip0.2em
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \hspace*{1em}{\color{myblue}\bfseries 定義}：直前の向きが次に\alert{影響}
      \begin{itemize}\footnotesize
        \item $P(\text{左}\mid\text{直前左}){=}a,\ P(\text{右}\mid\text{直前左}){=}c{=}1{-}a$
        \item $P(\text{左}\mid\text{直前右}){=}b,\ P(\text{右}\mid\text{直前右}){=}d{=}1{-}b$
      \end{itemize}
      \vskip0.3em
      コイン行列と\alert{左右移動への分解}
      \[ A=\begin{pmatrix}a&b\\c&d\end{pmatrix}
         =\underbrace{\begin{pmatrix}a&b\\0&0\end{pmatrix}}_{P\ (\text{左移動})}
          +\underbrace{\begin{pmatrix}0&0\\c&d\end{pmatrix}}_{Q\ (\text{右移動})} \]
    \end{column}
    \begin{column}{0.49\textwidth}
      {\color{myblue}\bfseries 確率ベクトルの時間発展}
      \begin{itemize}\footnotesize
        \item 時刻 $n$・位置 $x$ の確率ベクトルと初期値
          \[ \Psi_n(x)=\begin{pmatrix}\Psi_n^{L}(x)\\[1pt]\Psi_n^{R}(x)\end{pmatrix},\ \
             \Psi_0(0)=\varphi=\begin{pmatrix}\alpha\\\beta\end{pmatrix} \]
          \[ \alpha+\beta=1,\qquad \alpha,\beta\geq0 \]
        \item 発展則（$P$ が左へ，$Q$ が右へ運ぶ）
          \[ \Psi_{n+1}(x)=P\,\Psi_n(x{+}1)+Q\,\Psi_n(x{-}1) \]
        \item 位置の確率は成分の和
          \[ P(\tilde S_n{=}x)=\lVert\Psi_n(x)\rVert_1=\Psi_n^{L}(x)+\Psi_n^{R}(x) \]
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
\note{% 【約40秒】
相関付きランダムウォークです。通常のランダムウォークは各ステップが独立で，直前の向きは次に影響しませんが，このモデルでは直前の向きで確率を変えます。直前が左なら次に左へ行く確率を $a$，右なら右へ行く確率を $d$ と定め，コイン行列 $A$ にまとめます。ここが本質で，$A$ を左へ動かす成分 $P$ と右へ動かす成分 $Q$ に分解します。初期ベクトルの $\alpha$ と $\beta$ は左右それぞれの初期確率で，和は1です。あとは，確率ベクトル $\Psi$ に $P$ と $Q$ を掛けながら1歩ずつ時間発展させれば分布が求まります。位置の確率は成分の和です。なお，計算した例は付録に載せています。
\sourceblock{docs/thesis/main.tex, 第4章「相関付きランダムウォークについて」。}%
}

% =================  3. 確率分布で見る通常RWとの違い  ==============
\begin{frame}[t]{確率分布で見る通常のランダムウォークとの違い}
  \small
  \centering
  \safeincludegraphics[width=0.80\linewidth,height=0.645\textheight,keepaspectratio]{figures/crw_hist_n100}
  \par\vskip0.1em
  {\scriptsize\color{sub}時間発展対称な場合（$a{=}d$，$\varphi{=}(1/2,1/2)^{\top}$）の分布を
  $u{=}\tilde S_n/\sqrt{n}$ に再スケール。破線 $N(0,1)$ は\alert{通常のランダムウォークの極限}（極限分散 $\sigma^{2}{=}a/(1-a)$）}
  \vskip0.15em
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \begin{itemize}\footnotesize
        \item 1歩の上下の確率は\alert{どれも $1/2$}。違いは\alert{相関}だけ
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}\footnotesize
        \item 持続性：\alert{裾が広い}／反転傾向：\alert{原点に集中}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
\note{% 【約35秒】
この相関が分布にどう効くかを見ます。図は，時刻 $n$ が100のときの位置の分布を，$\sqrt{n}$ で割って再スケールしたものです。どちらも1歩あたり上下に動く確率は2分の1で同じで，違うのは直前の向きとの相関だけです。破線は標準正規分布で，これが相関のない通常のランダムウォークの極限にあたります。同じ向きが続きやすい持続性の場合は，一方向に長く進むので分布は破線より横に広がり，反転傾向の場合は行ったり来たりするので原点付近に集中します。
\sourceblock{docs/thesis/main.tex, 第4章「相関付きランダムウォークについて」（弱収束極限定理と分布の比較）。図は docs/thesis/crw\_hist\_n100.png（colab/oyosuri\_limit\_figs.py で生成）。}%
}

% =================  5. 予測モデル（評価関数・当てはめ）  ==========
\begin{frame}[t]{\hspace*{0.1em}相関付きRWによる時系列予測：評価関数と当てはめ}
  \small
  観測時系列 $x_0,\dots,x_n$（増分 $\pm1$，$x_0{=}0$）に対し，パラメータ
  $(p,q,\alpha)$ を推定して次時刻を予測する。
  \par{\footnotesize\color{mygray}$p,q$：左右の継続確率，$\alpha$：初期が左向きの確率}
  \vskip0.1em
  \centering{\color{myblue}\bfseries 実験で用いるコイン行列}\par
  \vskip-0.4em
  {\footnotesize
  \[
    A(p,q)=\begin{pmatrix}q&1-p\\1-q&p\end{pmatrix},
    \qquad
    \varphi(\alpha)=\begin{pmatrix}\alpha\\1-\alpha\end{pmatrix}
  \]}
  \vskip-0.4em
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      {\color{myblue}\bfseries ① モデル分布とのズレを測る}
      \begin{itemize}\footnotesize
        \item $\mu_t(x;p,q,\alpha){=}P(\tilde S_t{=}x)$
          \\ ＝\alert{発展則で求めるモデルの分布}そのもの
        \item 評価関数（二乗誤差を確率で重み付け）
          {\footnotesize
          \[V_n(p,q,\alpha){=}\sum_{t=0}^{n}\sum_{x}(x{-}x_t)^{2}\,\mu_t(x;p,q,\alpha)\]}
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      {\color{myblue}\bfseries ② パラメータを決めて予測する}
      \begin{itemize}\footnotesize
        \item 評価関数 $V_n$ が\alert{最小となるパラメータ}
          \\ $(p^\ast,q^\ast,\alpha^\ast)$ を決める
        \item 次時刻を\alert{期待値}で予測
          {\footnotesize
          \[x_{n+1}^{\ast}{=}E(\tilde S_{n+1}\mid p^\ast,q^\ast,\alpha^\ast){=}\sum_x x\,\mu_{n+1}(x;\!\cdot)\]}
      \end{itemize}
    \end{column}
  \end{columns}
  \vskip0.2em
  \centering{\footnotesize\color{mygray}以上を各 $n$ で繰り返し，予測列 $x_1^{\ast},\dots,x_N^{\ast}$ を得る}
\end{frame}
\note{% 【約30秒】
では，この分布をどう時系列予測に使うかです。実験では，先ほどの $a,b,c,d$ を $p,q$ で表し，コイン行列を $A(p,q)$，初期ベクトルを $\alpha$ と $1-\alpha$ で定めます。左の $\mu$ は，発展則で求めるモデルの分布です。評価関数 $V_n$ は，観測値 $x_t$ とモデルが取りうる位置 $x$ の二乗誤差を，確率 $\mu$ で重み付けしたものです。これが最小となるパラメータを決め，次時刻の期待値を予測値とします。この計算を各時刻で繰り返して，予測列を得ます。
\sourceblock{docs/thesis/main.tex, 第5章「時系列モデル」。src/oyosuri/model.py・optimize.py・predict.py。}%
}

% =====================  5. 離散化  ================================
\begin{frame}[t]{金価格データの離散化：平均練行足}
  \small
  \centering
  \safeincludegraphics[width=0.97\linewidth]{figures/linechart_renko_compare}
  \par\vskip0.1em
  {\scriptsize\color{sub}変換例（左：変換前の価格系列，右：変換後の平均練行足）}\par
  \vskip0.2em
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.60\textwidth}
      \begin{itemize}\footnotesize
        \item 価格が $B$ ドル動くごとに1本（$b_i{=}\pm1$）生成し，基準価格を更新
        \item 符号を累積して $x_n=\sum_{k=1}^{n}b_k$ を得る
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}
      \begin{itemize}\footnotesize
        \item 主実験：XAUUSD 1分足 317本，$B{=}3$ ドル
        \item 平均練行足 $N{=}156$ 本
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
\note{% 【約35秒】
実際の金価格は連続的に変動するので，そのままではモデルに入れられません。左が変換前の価格系列，右が変換後の平均練行足です。基準価格から $B$ 以上上がればプラス1の足，下がればマイナス1の足とし，価格が $B$ 動くごとに1本生成します。こうして得た符号を累積すると，ボックス単位の整数時系列 $x_n$ になります。主実験ではXAUUSDの1分足317本に対して $B=3$ ドルとし，156本の平均練行足を得ました。この離散系列をモデルの入力として使います。
\sourceblock{docs/thesis/main.tex, 第6章「解析手法」。図は figures/linechart\_renko\_compare.png。src/oyosuri/renko.py。}%
}

% =================  6. 実験：全期間参照の結果  ====================
\begin{frame}[t]{実験：金価格 XAUUSD への適用（全期間参照）}
  \small
  {\footnotesize {\color{myblue}\bfseries 設定}：XAUUSD 1分足（$B{=}3$ ドル，$N{=}156$）。予測と実測の\alert{重ね描き}＋補助指標\alert{MAE}（・HIT）で評価}
  \vskip-0.2em
  {\footnotesize
  \[
    \mathrm{MAE}=\frac{1}{N}\sum_{n=1}^{N}\bigl\lvert x_n-x_n^{\ast}\bigr\rvert
    \qquad
    \text{\small 練行足は毎回 $\pm1$ 動く\ $\Rightarrow$\ 「変化なし」予測は $\mathrm{MAE}{=}1$（基準）}
  \]}
  \vskip-0.5em
  \centering
  \safeincludegraphics[width=0.90\linewidth,height=0.44\textheight,keepaspectratio]{figures/full-history-overlay}
  \par{\scriptsize\color{sub}実測値と全期間参照による予測値の重ね描き}
  \vskip0.3em
  \begin{itemize}\footnotesize
    \item 全期間参照：$\mathrm{MAE}{=}5.94$（基準の約6倍），$\mathrm{HIT}{=}0.43$ $\Rightarrow$ \alert{誤差が大きい}
    \item 原因：古い値動きまで参照するため，\alert{直近の傾向に合わない}
  \end{itemize}
\end{frame}
\note{% 【約50秒】
実験です。対象は金価格 XAUUSD の1分足，値幅3ドルで作った練行足156本。予測と実測の重ね描きを中心に評価し，補助指標として平均絶対誤差MAEを使います。MAEは式のとおり，実測値と予測値の差の絶対値を平均したものです。練行足は毎回プラスマイナス1動くので，直前の値をそのまま予測する「変化なし」予測のMAEは常に1になり，これが基準になります。図が，全期間を参照して得た予測値と実測値の重ね描きです。MAEは5.94と基準の約6倍，方向的中率HITも0.43で，誤差が大きくなりました。金価格は上昇や下降など傾向が途中で変わりますが，全期間参照では過去のデータすべてから一組のパラメータを決めるため，古い傾向まで予測に混ざってしまいます。
\sourceblock{docs/thesis/main.tex, 第7章「実験結果」，式 (MAE)。差し替え図は figures/full-history-overlay。}%
}

% =================  7. 改良：参照期間 W  ==========================
\begin{frame}[t]{改良：参照期間 $W$ の導入}
  \small
  {\footnotesize {\color{myblue}\bfseries 改良後の評価関数}（直近 $W$ 点のみ）：先頭を $0$ に揃えた
  $y_j=x_{n-W+1+j}-x_{n-W+1}$（$j{=}0,\dots,W{-}1$）に対して}
  \vskip-0.4em
  {\footnotesize
  \[
    V_{W-1}(p,q,\alpha)=\sum_{t=0}^{W-1}\sum_{x}(x-y_t)^{2}\,\mu_t(x;p,q,\alpha)\ \to\ \text{最小化},
    \qquad
    x_{n+1}^{\ast}=x_{n-W+1}+E(\tilde S_{W}\mid p^{\ast},q^{\ast},\alpha^{\ast})
  \]}
  \vskip-0.6em
  \centering
  \safeincludegraphics[width=0.86\linewidth,height=0.46\textheight,keepaspectratio]{figures/w_sweep_summary}
  \par{\scriptsize\color{sub}参照期間 $W$ に対する MAE（左，破線は基準 $\mathrm{MAE}{=}1$）と1ステップ当たり計算時間（右）。
  全期間参照は $\mathrm{MAE}{=}5.94$（前ページ）}
  \vskip0.15em
  \begin{itemize}\footnotesize
    \item $W$ を小さくすると\alert{MAE・計算時間がともに減少}（トレードオフなし）
    \item $W{=}2$ で最良：$\mathrm{MAE}{=}0.847$（基準 $1$ 未満は $W{=}2,3$ のみ）
  \end{itemize}
\end{frame}
\note{% 【約45秒】
そこで改良として，当てはめに使う観測を直近 $W$ 点に限ります。直近 $W$ 点を取り出し，先頭を0に揃えた系列 $y$ を作り，これに対して評価関数を最小化します。式のとおり，和をとる範囲が全期間から直近 $W$ 点だけに変わったものが，改良後の評価関数です。予測値は，参照期間の先頭の値を基準として，そこから $W$ ステップ後の期待値を足したものとします。図の左が $W$ に対するMAE，右が1ステップ当たりの計算時間です。$W$ を小さくするほどMAEも計算時間も減り，精度と計算量のトレードオフは見られませんでした。最小の $W=2$ でMAEは0.847となり，基準の1を下回ったのは $W=2$ と3だけでした。
\sourceblock{docs/thesis/main.tex, 第7章「参照期間の導入」「$W$ に対する感度分析」。図は figures/w\_sweep\_summary.png。}%
}

% =====================  8. モデルの限界  ==========================
\begin{frame}[t]{モデルの限界：局面転換への追従の遅れ}
  \small
  \centering
  \safeincludegraphics[width=0.83\linewidth]{figures/pat_up2down_w}
  \par{\scriptsize\color{sub}上昇→下降の局面（合成データ，$N=150$）}
  \vskip0.15em
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \begin{itemize}\footnotesize
        \item $W{=}60$ は転換点を過ぎても上昇の勢いを引きずり大きく乖離
          （\alert{$\mathrm{MAE}\,4.876$} vs $W{=}2$ の $0.892$）
        \item 機構：予測は\alert{参照期間の先頭から $W$ 歩後の期待値}で，
          \alert{現在位置・直前の向きを条件づけない}
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{itemize}\footnotesize
        \item $W{=}2$ は\alert{この条件では精度を確かめられた}
          （$\mathrm{MAE}\,0.847{<}1$：「変化なし」予測より小さい）
        \item ただし\alert{一般の精度の証明ではない}：履歴を使うほど誤差が増える構造の下での最良で，HIT は約 $0.5$
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
\note{% 【約55秒】
一方で，この結果には限界もあります。図は，上昇から下降へ転じる局面です。参照期間の大きい $W=60$ は，転換点を過ぎても上昇の勢いを引きずり，MAEは4.876と，$W=2$ の0.892の5倍以上に悪化しています。原因はモデルの構造です。この予測は，参照期間の先頭から $W$ 歩後の期待値を求めるだけで，現在位置や直前の向きで条件づけていません。だから $W$ が大きいほど起点が遠くなって誤差が増え，局面が変わっても $W$ 本ぶんの遅れが出ます。$W=2$ については，今回の条件，つまり金の練行足に対してはMAEが基準の1を下回っており，精度が確かめられたと言えます。ただしそれは，履歴を使うほど誤差が増えるこの構造のもとでの最良であって，一般に予測が当たることの証明ではありません。実際，HITは $W$ によらずおよそ0.5にとどまっています。
\sourceblock{docs/thesis/main.tex, 第8章「考察」。図は figures/pat\_up2down\_w.png。}%
}

% =====================  9. まとめ  ================================
\begin{frame}[t]{まとめと今後の課題}
  \small
  \begin{block}{まとめ}
    \begin{itemize}
      \item 相関付きランダムウォークを，金価格 XAUUSD に\alert{平均練行足}を通して初めて適用
      \item 全期間参照では誤差が大きい（$\mathrm{MAE}=5.94$）。\alert{参照期間 $W$} の導入で改善し，$W=2$ で最良（$\mathrm{MAE}=0.847$：この条件では基準 $1$ を下回る）
      \item \alert{現在位置・向きを予測に使わないことが限界}であり，$W$ を大きくすると誤差が増える
    \end{itemize}
  \end{block}
  \begin{block}{今後の課題}
    \begin{itemize}\footnotesize
      \item 現在位置・直前の向きを\alert{条件づける}予測への拡張（条件付き期待値）
      \item 金以外の資産（株・為替・暗号資産・他商品）への適用
      \item 平均練行足に依らない離散化・非一定増分への拡張
      \item マルコフ連鎖・高次モデル・レジーム転換モデルとの比較
    \end{itemize}
  \end{block}
\end{frame}
\note{% 【約35秒】

\begin{frame}[t]{付録：相関付きRWの行列計算（$n=1,2$）}
  \small
  {\color{myblue}\bfseries $n=1$（原点から1歩目）}：初期ベクトル $\varphi$ に $P,Q$ を掛ける
  {\footnotesize
  \[\begin{aligned}
    \Psi_1(-1)&=P\varphi=\begin{pmatrix}a&b\\0&0\end{pmatrix}\begin{pmatrix}\alpha\\\beta\end{pmatrix}
      =\begin{pmatrix}a\alpha+b\beta\\0\end{pmatrix},\qquad
    \Psi_1(+1)=Q\varphi=\begin{pmatrix}0&0\\c&d\end{pmatrix}\begin{pmatrix}\alpha\\\beta\end{pmatrix}
      =\begin{pmatrix}0\\c\alpha+d\beta\end{pmatrix}
  \end{aligned}\]}
  {\color{myblue}\bfseries $n=2$（発展則 $\Psi_{n+1}(x){=}P\Psi_n(x{+}1){+}Q\Psi_n(x{-}1)$ を適用）}
  {\footnotesize
  \[\begin{aligned}
    \Psi_2(-2)&=P\Psi_1(-1)=\begin{pmatrix}a(a\alpha+b\beta)\\0\end{pmatrix},\qquad
    \Psi_2(+2)=Q\Psi_1(+1)=\begin{pmatrix}0\\d(c\alpha+d\beta)\end{pmatrix}\\[2pt]
    \Psi_2(0)&=P\Psi_1(+1)+Q\Psi_1(-1)
      =\begin{pmatrix}b(c\alpha+d\beta)\\0\end{pmatrix}+\begin{pmatrix}0\\c(a\alpha+b\beta)\end{pmatrix}
      =\begin{pmatrix}b(c\alpha+d\beta)\\c(a\alpha+b\beta)\end{pmatrix}
  \end{aligned}\]}
  \centering{\footnotesize\color{mygray}位置の確率は成分の和：$P(\tilde S_2{=}0)=\lVert\Psi_2(0)\rVert_1=b(c\alpha+d\beta)+c(a\alpha+b\beta)$}
\end{frame}
\note{% 【付録・質疑用】
1歩目は，初期ベクトル $\varphi$ に $P$ と $Q$ を掛けるだけです。$P$ を掛けると左向き成分に $a\alpha{+}b\beta$，$Q$ を掛けると右向き成分に $c\alpha{+}d\beta$ が現れ，これがそれぞれ位置マイナス1，プラス1にいる確率です。2歩目は，発展則にしたがって1歩目のベクトルにまた $P$ と $Q$ を掛けて足します。たとえば原点に戻る $\Psi_2(0)$ は，右にいた状態に $P$ を掛けたものと，左にいた状態に $Q$ を掛けたものの和になります。$n=3$ 以降は経路の順序ごとに行列積が増えていきます。
\sourceblock{docs/thesis/main.tex, 第4章の \(n=1,2\) の時間発展および経路和。}%
}
